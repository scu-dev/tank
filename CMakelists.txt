cmake_minimum_required(VERSION "3.22")

project("tank")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

cmake_policy(SET CMP0091 NEW)
if(MSVC)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /Zc:__cplusplus /Zc:preprocessor")
    add_compile_definitions(
        WIN32_LEAN_AND_MEAN
        NOMINMAX
    )
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    foreach(flag_var
        CMAKE_C_FLAGS CMAKE_C_FLAGS_DEBUG CMAKE_C_FLAGS_RELEASE
        CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE)
        if(${flag_var} MATCHES "/MD")
            string(REGEX REPLACE "/MD" "/MT" ${flag_var} "${${flag_var}}")
        endif()
    endforeach()
endif()

set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE INTERNAL "")

file(GLOB_RECURSE TANK_SOURCES CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.cpp")
file(GLOB_RECURSE IMGUI_SOURCES CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/libs/imgui/*.cpp")

set(SDL_SHARED        OFF CACHE BOOL "")
set(SDL_STATIC        ON  CACHE BOOL "")
set(SDL_EXAMPLES      OFF CACHE BOOL "")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "")
add_subdirectory("libs/sdl")
set(SDLIMAGE_AVIF     OFF CACHE BOOL "")
set(SDLIMAGE_BMP      OFF CACHE BOOL "")
set(SDLIMAGE_GIF      OFF CACHE BOOL "")
set(SDLIMAGE_JPG      OFF CACHE BOOL "")
set(SDLIMAGE_JXL      OFF CACHE BOOL "")
set(SDLIMAGE_LBM      OFF CACHE BOOL "")
set(SDLIMAGE_PCX      OFF CACHE BOOL "")
set(SDLIMAGE_PNM      OFF CACHE BOOL "")
set(SDLIMAGE_QOI      OFF CACHE BOOL "")
set(SDLIMAGE_SVG      OFF CACHE BOOL "")
set(SDLIMAGE_TGA      OFF CACHE BOOL "")
set(SDLIMAGE_TIF      OFF CACHE BOOL "")
set(SDLIMAGE_WEBP     OFF CACHE BOOL "")
set(SDLIMAGE_XCF      OFF CACHE BOOL "")
set(SDLIMAGE_XPM      OFF CACHE BOOL "")
set(SDLIMAGE_XV       OFF CACHE BOOL "")
add_subdirectory("libs/sdl_image")

add_executable(${PROJECT_NAME} ${TANK_SOURCES} ${IMGUI_SOURCES})

target_link_options(${PROJECT_NAME} PRIVATE "/SUBSYSTEM:WINDOWS")

target_include_directories(${PROJECT_NAME} PRIVATE
    "libs/sdl/include"
    "libs/sdl_image/include"
	"libs/imgui"
    "${CMAKE_SOURCE_DIR}/packing"
)

target_sources(${PROJECT_NAME} PRIVATE "${CMAKE_SOURCE_DIR}/packing/tank.rc")

target_link_libraries(${PROJECT_NAME} PRIVATE
    SDL3-static
    SDL3_image-static
)

if(EXISTS "${CMAKE_SOURCE_DIR}/images")
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/images"  "$<TARGET_FILE_DIR:${PROJECT_NAME}>/images")
endif()

if(EXISTS "${CMAKE_SOURCE_DIR}/sounds")
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/sounds"  "$<TARGET_FILE_DIR:${PROJECT_NAME}>/sounds")
endif()

if(EXISTS "${CMAKE_SOURCE_DIR}/map")
    add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/map"  "$<TARGET_FILE_DIR:${PROJECT_NAME}>/map")
endif()
